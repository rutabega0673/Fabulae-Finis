country_decisions = {
	become_cascadian_diarchy_tribal = {
		potential = {
			num_of_cities >= 1
			OR = { 
				religion = gaian
				government = tribal_federation
				government = tribal_kingdom
			}
			is_ai = yes
		}
		allow = {
			ai_is_allowed_to_reform_tribe = yes
			has_law = absolute_authority
			NOT = {
				any_character = {
					is_clan_chief = yes
					loyalty < 33
				}
			}
		}
		effect = {
			add_political_influence = -100
			trigger_event = {
				id = nation_forming.3
			}
		}
		
		ai_will_do = {
			factor = 1000
			modifier = {
				factor = 0
				war = yes
			}
		}
	}
	become_cascadian_diarchy_monarchy = {
		potential = {
			num_of_cities >= 1
			OR = {
				government = despotic_monarchy
				government = aristocratic_monarchy
				government = stratocratic_monarchy
				government = plutocratic_monarchy
			}
		}
		allow = {
			stability >= 40
			capital_scope = { civilization_value >= 30 }
			religious_tech >= 4
		}
		effect = {
			if = {
				limit = {
					OR = {
						government = aristocratic_monarchy
						government = stratocratic_monarchy
						government = plutocratic_monarchy
					}
				}
				add_stability = -20
				change_government = cascadian_diarchy
			}
			else = {
				add_stability = -10
				change_government = cascadian_diarchy
			}
			switch_government_type_event_clearup_effect = yes
		}
		
		ai_will_do = {
			factor = 1
			modifier = {
				factor = 0
				NOT = {
					current_ruler = {
						char_is_decent_trigger = yes
					}
				}
			}
			modifier = {
				factor = 0
				OR = {
					government = stratocratic_monarchy
					government = plutocratic_monarchy
					government = aristocratic_monarchy
				}
			}
		}
	}
}
